(function() {
  const STORAGE_KEY = 'uppcs_thematic_tracker_v1';
  
  function initTracker() {
    const table = document.querySelector('#thematic-master-table table');
    if (!table) return;

    let saved = {};
    try {
      saved = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
    } catch (e) {}

    const rows = table.querySelectorAll('tbody tr');
    let totalChecks = 0;
    let checkedCount = 0;

    rows.forEach((row, rIdx) => {
      const checkboxes = row.querySelectorAll('input[type="checkbox"]');
      checkboxes.forEach((cb, cIdx) => {
        cb.disabled = false;
        cb.style.cursor = 'pointer';
        cb.style.transform = 'scale(1.2)';
        const key = `t_${rIdx}_${cIdx}`;
        if (saved[key]) {
          cb.checked = true;
          checkedCount++;
        }
        totalChecks++;

        cb.addEventListener('change', () => {
          saved[key] = cb.checked;
          localStorage.setItem(STORAGE_KEY, JSON.stringify(saved));
          updateStats();
        });
      });
    });

    function updateStats() {
      let checked = 0;
      rows.forEach((row, rIdx) => {
        const cbs = row.querySelectorAll('input[type="checkbox"]');
        cbs.forEach((cb, cIdx) => {
          if (cb.checked) checked++;
        });
      });
      const pct = Math.round((checked / (totalChecks || 1)) * 100);
      
      const statDone = document.getElementById('stat-done-checks');
      const statPct = document.getElementById('stat-percent');
      const progLabel = document.getElementById('progress-label');
      const progBar = document.getElementById('progress-bar-fill');
      
      if (statDone) statDone.textContent = `${checked} / ${totalChecks}`;
      if (statPct) statPct.textContent = `${pct}%`;
      if (progLabel) progLabel.textContent = `${pct}%`;
      if (progBar) progBar.style.width = `${pct}%`;
    }

    updateStats();
  }

  window.filterThematic = function(sec) {
    const table = document.querySelector('#thematic-master-table table');
    if (!table) return;
    const rows = table.querySelectorAll('tbody tr');
    rows.forEach(row => {
      if (sec === 'all') {
        row.style.display = '';
      } else {
        const code = row.querySelector('code');
        if (code && code.textContent.trim() === sec) {
          row.style.display = '';
        } else {
          row.style.display = 'none';
        }
      }
    });
  };

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initTracker);
  } else {
    initTracker();
  }
})();
</script>
